## 1.4 核心业务逻辑

### 1.4.1 政策生效规则

```
1. 按组织匹配：优先匹配具体组织 → 上级组织 → 全局
2. 按时间匹配：当前日期在生效区间内
3. 按优先级匹配：多个匹配时取优先级最高的
4. 按费用类型匹配：精确匹配
5. 多组织支持：一个政策可同时适用多个组织
```

### 1.4.2 审批路由计算逻辑

```python
def calculate_approval_route(expense_type, amount, org_id, employee_level,
                             is_overseas, is_overtime, extra_conditions):
    # 1. 获取适用的费用政策
    policy = find_applicable_policy(expense_type, org_id)
    if not policy:
        return get_default_approval_route(expense_type, amount)
    
    # 2. 获取政策下的审批规则
    rules = policy.get_approval_rules()
    
    # 3. 按规则匹配审批路径
    approval_route = []
    for rule in rules:
        if match_rule(rule, amount, employee_level, is_overseas,
                     is_overtime, extra_conditions):
            approval_route.append({
                'level': rule.auth_level,
                'role_id': rule.auth_role_id,
                'priority': rule.priority
            })
    
    # 4. 添加附加审批节点
    if is_overtime:
        approval_route.append({
            'level': policy.get_overtime_rule().level,
            'type': 'overtime_approval'
        })
    
    if extra_conditions.get('project_bearing'):
        approval_route.append({'level': 'PROJECT_MANAGER', 'type': 'project_approval'})
    
    # 5. 排序去重
    approval_route.sort(key=lambda x: x['priority'], reverse=True)
    return deduplicate_route(approval_route)
```

### 1.4.3 差旅标准计算

```python
def calculate_travel_standard(employee_level, origin_city, dest_city, days):
    # 1. 获取员工等级对应的差旅标准
    standard = get_travel_standard(employee_level)
    
    # 2. 获取城市等级系数
    avg_factor = (get_tier_factor(get_city_tier(origin_city)) +
                  get_tier_factor(get_city_tier(dest_city))) / 2
    
    # 3. 判断境内/境外
    is_international = is_international_travel(origin_city, dest_city)
    
    # 4. 计算各项费用
    if is_international:
        accommodation = standard.international_accommodation * avg_factor * days
        meal = standard.international_meal * avg_factor * days
    else:
        accommodation = standard.domestic_accommodation * avg_factor * days
        meal = standard.domestic_meal * avg_factor * days
    
    return {
        'accommodation': round(accommodation, 2),
        'meal': round(meal, 2),
        'total': round(accommodation + meal, 2),
        'city_factor': avg_factor
    }
```

### 2.5.1 预检执行引擎

```python
class PrecheckEngine:
    """合规预检引擎"""
    
    def execute(self, bill_data):
        """执行全量预检"""
        batch_id = self._create_batch(bill_data)
        results = []
        
        # 1. 获取适用规则并分组
        applicable_rules = self._filter_rules(bill_data)
        parallel_rules = [r for r in applicable_rules if r.is_parallel]
        serial_rules = [r for r in applicable_rules if not r.is_parallel]
        
        # 2. 并行执行
        with ThreadPoolExecutor(max_workers=5) as executor:
            futures = {executor.submit(self._execute_rule, r, bill_data): r 
                      for r in parallel_rules}
            for future in as_completed(futures):
                result = future.result()
                results.append(result)
                self._save_result(batch_id, result)
        
        # 3. 串行执行（遇ERROR停止）
        for rule in serial_rules:
            result = self._execute_rule(rule, bill_data)
            results.append(result)
            if result.level == 'ERROR':
                break
        
        # 4. 汇总状态
        final_status = self._calculate_status(results)
        self._update_batch(batch_id, results, final_status)
        
        return PrecheckResult(batch_id, results, final_status)
    
    def _execute_rule(self, rule, bill_data):
        """执行单条规则"""
        start_time = time.time()
        try:
            params = self._prepare_params(rule.input_params, bill_data)
            result = self._eval_script(rule.script_content, params)
            return {
                'rule_code': rule.rule_code,
                'result': result.get('result', 'SKIP'),
                'message': result.get('message', ''),
                'level': result.get('level', 'INFO'),
                'execution_time_ms': int((time.time() - start_time) * 1000)
            }
        except Exception as e:
            return {
                'rule_code': rule.rule_code,
                'result': 'FAIL',
                'message': f'规则执行异常: {str(e)}',
                'level': 'ERROR',
                'execution_time_ms': int((time.time() - start_time) * 1000)
            }
```

```python
async def verify_invoice(code, number, date=None, amount=None):
    """调用税务接口验证发票真伪"""
    cache_key = f"invoice:{code}:{number}"
    cached = await redis.get(cache_key)
    if cached:
        return json.loads(cached)
    
    try:
        response = await tax_service.verify_invoice({
            "invoiceCode": code,
            "invoiceNumber": number,
            "invoiceDate": date,
            "amountWithoutTax": amount
        })
        result = {
            "valid": response.get("code") == "0000",
            "message": response.get("message", "")
        }
        await redis.setex(cache_key, 604800, json.dumps(result))
        return result
    except Exception as e:
        logger.warning(f"Invoice verification failed: {e}")
```

```python
class SmartDispatcher:
    """智能派单引擎"""
    
    def dispatch(self, bill_data):
        """执行派单"""
        # 1. 计算复杂度评分
        complexity = self._calculate_complexity(bill_data)
        
        # 2. 匹配派单规则
        rule = self._match_rule(bill_data, complexity)
        
        # 3. 选择目标审核组
        target_group = rule.target_group
        
        # 4. 在组内选择审核员
        reviewer = self._select_reviewer(target_group, rule.dispatch_strategy)
        
        # 5. 创建派单任务
        task = DispatchTask(
            bill_id=bill_data.id,
            expense_type=bill_data.expense_type,
            amount=bill_data.amount,
            complexity_score=complexity,
            assigned_group_id=target_group.id,
            assigned_reviewer_id=reviewer.id,
            due_time=self._calculate_due_time(bill_data)
        )
        
        return task
    
    def _match_rule(self, bill_data, complexity):
        """匹配派单规则"""
        rules = self.get_active_rules()
        
        for rule in sorted(rules, key=lambda r: r.priority, reverse=True):
            if self._evaluate_condition(rule.match_condition, bill_data, complexity):
                return rule
        
        return self.get_default_rule()
    
    def _select_reviewer(self, group, strategy):
        """按策略选择审核员"""
        available = [r for r in group.reviewers 
                    if r.status == 'active' and r.current_pending < r.max_daily_tasks]
        
        if strategy == 'ROUND_ROBIN':
            return self._round_robin_select(group, available)
        elif strategy == 'LOAD_BALANCE':
            return min(available, key=lambda r: r.current_pending)
        elif strategy == 'SKILL_MATCH':
            return self._skill_match_select(available)
        else:
            return random.choice(available)
```

### 3.4.2 复杂度评分算法

```python
def calculate_complexity_score(bill_data):
    """计算单据复杂度（0-10分）"""
    score = 0
    
    # 1. 金额维度（0-3分）
    amount = bill_data.amount
    if amount > 500000:
        score += 3
    elif amount > 100000:
        score += 2
    elif amount > 50000:
        score += 1
    
    # 2. 发票数量（0-2分）
    invoice_count = len(bill_data.invoices)
    if invoice_count > 20:
        score += 2
    elif invoice_count > 10:
        score += 1
    
    # 3. 涉及多方（0-2分）
    if bill_data.is_related_to_contract:
        score += 1
    if bill_data.is_related_to_project:
        score += 1
    
    # 4. 跨法人/跨区域（0-2分）
    if bill_data.cross_entity:
        score += 1
    if bill_data.international:
        score += 1
    
    # 5. 历史异常（0-1分）
    if bill_data.has_history_warnings:
        score += 1
    
    return min(score, 10)
```

```python
def verify_integrity(self, archive_id):
    """校验档案完整性"""
    archive = archive_repository.get(archive_id)
    attachments = attachment_repository.get_by_archive(archive_id)
    
    # 1. 重新计算哈希
    new_hash = self._calculate_integrity_hash(archive, attachments)
    
    # 2. 对比原哈希
    if new_hash == archive.integrity_hash:
        archive.integrity_status = 'VERIFIED'
    else:
        archive.integrity_status = 'MISMATCH'
        # 发送告警
        self._alert_integrity_issue(archive)
    
    archive_repository.save(archive)
    return archive.integrity_status

def _calculate_integrity_hash(self, archive, attachments):
    """计算完整性哈希"""
    hasher = hashlib.sha256()
    
    # 归档基本信息
    hasher.update(json.dumps({
        'archive_code': archive.archive_code,
        'archive_type': archive.archive_type,
        'bill_id': archive.bill_id,
        'voucher_id': archive.voucher_id
    }).encode())
    
    # 所有附件的哈希
    for att in sorted(attachments, key=lambda a: a.id):
        hasher.update(f"{att.file_name}:{att.content_hash}".encode())
    
    return hasher.hexdigest()
```
